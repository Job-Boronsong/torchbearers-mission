server {
    server_name torchbearersmission.org www.torchbearersmission.org;

    client_max_body_size 20M;

    # Serve React frontend static build
    location / {
        root /var/www/torchbearers/app/frontend/dist;
        try_files $uri $uri/ /index.html;
    }

    # Proxy API and admin to Django/Gunicorn
    location /api/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /admin/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /media/ {
        alias /var/www/torchbearers/app/media/;
    }

    location /static/ {
        alias /var/www/torchbearers/app/staticfiles/;
    }

    location /ckeditor/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    location /ckeditor5/ {
        proxy_pass http://127.0.0.1:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/torchbearersmission.org-0001/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/torchbearersmission.org-0001/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}
server {
    if ($host = www.torchbearersmission.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = torchbearersmission.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name torchbearersmission.org www.torchbearersmission.org;
    return 404; # managed by Certbot
